    Odrediti napon $V_{out}$ za kolo sa slike \ref{fig:zad1}, ako su $R_1 = 4 \,k\,\Omega$ i $R_2 = 6\,k\,\Omega$, a
    \begin{enumerate}
        \item [a)] $V_{in} = 10 \,V$
        \item [b)] $V_{in} = 10\sin{(20\pi\,t)} \,V$
    \end{enumerate}
\end{zadatak}
