\section{Zadaci}

\newtheorem{zadatak}{Zadatak}

\begin{zadatak}
    Sastaviti kolo neinvertujućeg operacionog pojačavača, koristiti operacioni pojačavač LT1014, gde su $\pm V_{cc}=\pm15\,V$, a $R_1 = 1 \,kV$ i $R_2 = 1 \,MV$, i odrediti:

    \begin{enumerate}
        \item [a)] Vrednost struja polarizacije za kolo,
        \item [b)] Izlaz iz operacionog pojačavača za proizvoljnu vrednost $V_{in}$,
        \item [c)] Eliminisati uticaj struja polarizacije na izlaz kola.
    \end{enumerate}
\end{zadatak}

\begin{zadatak}
    Sastaviti kolo neinvertujućeg operacionog pojačavača, koristiti operacioni pojačavač LT1014, gde su $\pm V_{cc}=\pm15\,V$, a $R_1 = 1 \,kV$ i $R_2 = 1 \,MV$, i odrediti:
    
    \begin{enumerate}
        \item [a)] Vrednost naponskog ofseta u kolu,
        \item [b)] Izlaz iz operacionog pojačavača za proizvoljnu vrednost $V_{in}$,
        \item [c)] Eliminisati uticaj naponskog ofseta na izlaz kola.
    \end{enumerate}
\end{zadatak}

\begin{zadatak}
    Sastaviti kolo neinvertujućeg operacionog pojačavača, koristiti Universal Op Amp 2, gde su $\pm V_{cc}=\pm15\,V$, a $R_1 = 1 \,kV$ i $R_2 = 1 \,MV$, i odrediti:
    
    \begin{enumerate}
        \item [a)] bodeov diagram pojačanja operacionog pojačavača u otvorenoj sprezi sa ulaznim AC u vrednosti od 1
        \item [b)] bodeov diagram pojačanja operaciongo pojačavača u zatvorenoj sprezi
    \end{enumerate}
\end{zadatak}
